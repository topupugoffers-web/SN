// ============================================================================
// CONFIRMED HTF TREND
// ============================================================================

f_confirmedBullTrend(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         (
             ta.ema(close, emaFastLen)[1] >
             ta.ema(close, emaSlowLen)[1] and
             close[1] >
             ta.ema(close, emaFastLen)[1]
         ),
         barmerge.gaps_off,
         barmerge.lookahead_on)


f_confirmedBearTrend(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         (
             ta.ema(close, emaFastLen)[1] <
             ta.ema(close, emaSlowLen)[1] and
             close[1] <
             ta.ema(close, emaFastLen)[1]
         ),
         barmerge.gaps_off,
         barmerge.lookahead_on)


// ============================================================================
// CONFIRMED HTF STRUCTURE
// ============================================================================

f_confirmedStructureBull(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         close[1] > ta.highest(high[2], swingLen * 2),
         barmerge.gaps_off,
         barmerge.lookahead_on)


f_confirmedStructureBear(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         close[1] < ta.lowest(low[2], swingLen * 2),
         barmerge.gaps_off,
         barmerge.lookahead_on)


// ============================================================================
// CONFIRMED HTF LOCATION
// ============================================================================

f_confirmedLocationBull(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         close[1] > ta.ema(close, emaFastLen)[1],
         barmerge.gaps_off,
         barmerge.lookahead_on)


f_confirmedLocationBear(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         close[1] < ta.ema(close, emaFastLen)[1],
         barmerge.gaps_off,
         barmerge.lookahead_on)


// ============================================================================
// CONFIRMED HTF LIQUIDITY
// ============================================================================

f_confirmedSweepBull(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         low[1] < ta.lowest(low[2], liquidityLookback) and
         close[1] > ta.lowest(low[2], liquidityLookback),
         barmerge.gaps_off,
         barmerge.lookahead_on)


f_confirmedSweepBear(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         high[1] > ta.highest(high[2], liquidityLookback) and
         close[1] < ta.highest(high[2], liquidityLookback),
         barmerge.gaps_off,
         barmerge.lookahead_on)


// ============================================================================
// HTF DATA
// ============================================================================

biasBull =
     f_confirmedBullTrend(biasTF)

biasBear =
     f_confirmedBearTrend(biasTF)

structureBull =
     f_confirmedStructureBull(structureTF)

structureBear =
     f_confirmedStructureBear(structureTF)

locationBull =
     f_confirmedLocationBull(locationTF)

locationBear =
     f_confirmedLocationBear(locationTF)

locationSweepBull =
     f_confirmedSweepBull(locationTF)

locationSweepBear =
     f_confirmedSweepBear(locationTF)


// ============================================================================
// CURRENT TIMEFRAME
// ============================================================================

atr = ta.atr(atrLen)

atrAverage = ta.sma(atr, atrAverageLen)

atrRatio =
     atrAverage > 0 ?
     atr / atrAverage :
     1.0

volatilityOK =
     not useVolatilityFilter or
     (
         atrRatio >= minATRFactor and
         atrRatio <= maxATRFactor
     )

emaFast = ta.ema(close, emaFastLen)

emaSlow = ta.ema(close, emaSlowLen)


// ============================================================================
// CANDLE DATA
// ============================================================================

bull = close > open

bear = close < open

body =
     math.abs(close - open)

upperWick =
     high - math.max(open, close)

lowerWick =
     math.min(open, close) - low

bodyOK =
     body >= atr * minBodyATR


// ============================================================================
// PIN BARS
// ============================================================================

bullPin =
     usePinBar and
     bull and
     bodyOK and
     lowerWick >= body * pinWickRatio and
     upperWick <= body

bearPin =
     usePinBar and
     bear and
     bodyOK and
     upperWick >= body * pinWickRatio and
     lowerWick <= body


// ============================================================================
// ENGULFING
// ============================================================================

bullEngulf =
     useEngulfing and
     bull and
     bear[1] and
     open <= close[1] and
     close >= open[1]

bearEngulf =
     useEngulfing and
     bear and
     bull[1] and
     open >= close[1] and
     close <= open[1]


bullPA =
     bullPin or bullEngulf

bearPA =
     bearPin or bearEngulf


// ============================================================================
// CURRENT-TF LIQUIDITY
// ============================================================================

previousLow =
     ta.lowest(low[1], liquidityLookback)

previousHigh =
     ta.highest(high[1], liquidityLookback)

bullSweep =
     low < previousLow and
     close > previousLow and
     (previousLow - low) <= atr * sweepATR

bearSweep =
     high > previousHigh and
     close < previousHigh and
     (high - previousHigh) <= atr * sweepATR


// Store the actual sweep extreme.
// This becomes the structural invalidation reference.

var float lastBullSweepLow = na
var float lastBearSweepHigh = na

if bullSweep
    lastBullSweepLow := low

if bearSweep
    lastBearSweepHigh := high


// ============================================================================
// CURRENT-TF BOS
// ============================================================================

recentHigh =
     ta.highest(high[1], bosLookback)

recentLow =
     ta.lowest(low[1], bosLookback)

bullBOS =
     close > recentHigh

bearBOS =
     close < recentLow


// ============================================================================
// LOCAL TREND
// ============================================================================

localBull =
     emaFast > emaSlow and
     close > emaFast

localBear =
     emaFast < emaSlow and
     close < emaFast


// ============================================================================
// LOCATION ALIGNMENT
// ============================================================================

locationBullOK =
     not useLocation or
     locationBull or
     locationSweepBull

locationBearOK =
     not useLocation or
     locationBear or
     locationSweepBear

locationSweepBullOK =
     not requireLocationSweep or
     locationSweepBull

locationSweepBearOK =
     not requireLocationSweep or
     locationSweepBear


// ============================================================================
// SCORE
//
// Maximum = 11
//
// 2 = HTF bias
// 2 = HTF structure
// 1 = HTF location
// 2 = HTF liquidity
// 1 = 5M sweep
// 1 = price action
// 1 = BOS
// 1 = local trend
//
// Raw total can reach 11+ if every condition is present.
// The score is capped at 11 for display.
// ============================================================================

bullScoreRaw = 0

bullScoreRaw += biasBull ? 2 : 0
bullScoreRaw += structureBull ? 2 : 0
bullScoreRaw += locationBull ? 1 : 0
bullScoreRaw += locationSweepBull ? 2 : 0
bullScoreRaw += bullSweep ? 1 : 0
bullScoreRaw += bullPA ? 1 : 0
bullScoreRaw += bullBOS ? 1 : 0
bullScoreRaw += localBull ? 1 : 0

bullScore =
     math.min(bullScoreRaw, 11)


bearScoreRaw = 0

bearScoreRaw += biasBear ? 2 : 0
bearScoreRaw += structureBear ? 2 : 0
bearScoreRaw += locationBear ? 1 : 0
bearScoreRaw += locationSweepBear ? 2 : 0
bearScoreRaw += bearSweep ? 1 : 0
bearScoreRaw += bearPA ? 1 : 0
bearScoreRaw += bearBOS ? 1 : 0
bearScoreRaw += localBear ? 1 : 0

bearScore =
     math.min(bearScoreRaw, 11)


// ============================================================================
// SEQUENCE VALIDATION
//
// This is the main V3 improvement.
//
// BUY sequence:
//
// HTF alignment
// +
// liquidity sweep
// +
// rejection / PA
// +
// BOS
// +
// local trend
//
// SELL is opposite.
// ============================================================================

bullHTFAligned =
     (not useBias or biasBull) and
     (not useStructure or structureBull) and
     locationBullOK and
     locationSweepBullOK

bearHTFAligned =
     (not useBias or biasBear) and
     (not useStructure or structureBear) and
     locationBearOK and
     locationSweepBearOK


bullTrigger =
     bullSweep and
     bullPA and
     bullBOS

bearTrigger =
     bearSweep and
     bearPA and
     bearBOS


// ============================================================================
// STRUCTURAL STOP
// ============================================================================

bullStructuralStop =
     not na(lastBullSweepLow) ?
     lastBullSweepLow - atr * slATRBuffer :
     low - atr * slATRBuffer

bearStructuralStop =
     not na(lastBearSweepHigh) ?
     lastBearSweepHigh + atr * slATRBuffer :
     high + atr * slATRBuffer


// ============================================================================
// POTENTIAL ENTRY / RISK
// ============================================================================

potentialBuyEntry =
     close

potentialSellEntry =
     close

potentialBuySL =
     bullStructuralStop

potentialSellSL =
     bearStructuralStop

buyRisk =
     potentialBuyEntry - potentialBuySL

sellRisk =
     potentialSellSL - potentialSellEntry


buyTP2 =
     potentialBuyEntry + buyRisk * tp2RR

sellTP2 =
     potentialSellEntry - sellRisk * tp2RR


buyRR =
     buyRisk > 0 ?
     math.abs(buyTP2 - potentialBuyEntry) / buyRisk :
     0

sellRR =
     sellRisk > 0 ?
     math.abs(potentialSellEntry - sellTP2) / sellRisk :
     0


buyRiskOK =
     buyRisk > syminfo.mintick and
     buyRR >= minimumRR

sellRiskOK =
     sellRisk > syminfo.mintick and
     sellRR >= minimumRR


// ============================================================================
// COOLDOWN
// ============================================================================

var int lastSignalBar = na

cooldownOK =
     na(lastSignalBar) or
     bar_index - lastSignalBar > cooldownBars


// ============================================================================
// WATCH CONDITIONS
//
// WATCH means conditions are developing.
// It NEVER produces the confirmed BUY/SELL marker.
// ============================================================================

watchBuy =
     chartTFValid and
     inSession and
     volatilityOK and
     cooldownOK and
     bullHTFAligned and
     bullSweep and
     bullPA and
     not bullBOS

watchSell =
     chartTFValid and
     inSession and
     volatilityOK and
     cooldownOK and
     bearHTFAligned and
     bearSweep and
     bearPA and
     not bearBOS


// ============================================================================
// FINAL CONFIRMED SETUP
//
// ALL major sequence requirements must be satisfied.
//
// No signal until the chart candle closes.
// ============================================================================

buySignal =
     barstate.isconfirmed and
     chartTFValid and
     inSession and
     volatilityOK and
     cooldownOK and
     bullHTFAligned and
     (not requireSweep or bullSweep) and
     (not requirePA or bullPA) and
     (not requireBOS or bullBOS) and
     localBull and
     bullScore >= minimumScore and
     bullScore >= confirmedScore and
     buyRiskOK

sellSignal =
     barstate.isconfirmed and
     chartTFValid and
     inSession and
     volatilityOK and
     cooldownOK and
     bearHTFAligned and
     (not requireSweep or bearSweep) and
     (not requirePA or bearPA) and
     (not requireBOS or bearBOS) and
     localBear and
     bearScore >= minimumScore and
     bearScore >= confirmedScore and
     sellRiskOK

    lastDirection := "SELL"
