indicator("XAUUSD Smart Price Action • 5M/15M", overlay=true, max_lines_count=500, max_labels_count=500)

// ═══════════════════════════════════════════════════════════════
// XAUUSD SMART PRICE ACTION
// Liquidity Sweep → Price Action → BOS → Entry → ATR SL/TP
// ═══════════════════════════════════════════════════════════════

// ───────────────────────────────────────────────────────────────
// SETTINGS
// ───────────────────────────────────────────────────────────────

groupMode = "⚙️ Trading Mode"

preset = input.string(
     "5M",
     "Preset",
     options=["5M", "15M", "Custom"],
     group=groupMode)

useHTF = input.bool(true, "Use Higher-Timeframe Filter", group=groupMode)
htfTF  = input.timeframe("15", "Higher Timeframe", group=groupMode)

groupStructure = "📐 Market Structure"

swingLen = input.int(
     5,
     "Swing Length",
     minval=2,
     maxval=20,
     group=groupStructure)

bosLookback = input.int(
     10,
     "BOS Lookback",
     minval=3,
     maxval=30,
     group=groupStructure)

requireBOS = input.bool(
     true,
     "Require BOS Confirmation",
     group=groupStructure)

groupLiquidity = "💧 Liquidity Sweep"

sweepLookback = input.int(
     10,
     "Liquidity Lookback",
     minval=3,
     maxval=30,
     group=groupLiquidity)

sweepATR = input.float(
     0.15,
     "Maximum Sweep ATR",
     minval=0.05,
     maxval=1.0,
     step=0.05,
     group=groupLiquidity)

requireSweep = input.bool(
     true,
     "Require Liquidity Sweep",
     group=groupLiquidity)

groupPA = "🕯️ Price Action"

usePinBar = input.bool(
     true,
     "Use Pin Bars",
     group=groupPA)

useEngulfing = input.bool(
     true,
     "Use Engulfing",
     group=groupPA)

minBodyATR = input.float(
     0.08,
     "Minimum Candle Body / ATR",
     minval=0.01,
     maxval=0.5,
     step=0.01,
     group=groupPA)

wickRatio = input.float(
     2.0,
     "Pin Bar Wick / Body",
     minval=1.0,
     maxval=5.0,
     step=0.25,
     group=groupPA)

groupTrend = "📈 Trend Filter"

emaFastLen = input.int(
     20,
     "Fast EMA",
     minval=5,
     maxval=100,
     group=groupTrend)

emaSlowLen = input.int(
     50,
     "Slow EMA",
     minval=10,
     maxval=200,
     group=groupTrend)

useTrend = input.bool(
     true,
     "Require EMA Trend",
     group=groupTrend)

groupATR = "🎯 ATR / Risk"

atrLen = input.int(
     14,
     "ATR Length",
     minval=5,
     maxval=50,
     group=groupATR)

slATR = input.float(
     0.35,
     "SL ATR Buffer",
     minval=0.05,
     maxval=2.0,
     step=0.05,
     group=groupATR)

tp1RR = input.float(
     1.0,
     "TP1 Risk:Reward",
     minval=0.5,
     maxval=5.0,
     step=0.25,
     group=groupATR)

tp2RR = input.float(
     2.0,
     "TP2 Risk:Reward",
     minval=1.0,
     maxval=10.0,
     step=0.25,
     group=groupATR)

groupFilter = "🔎 Signal Filters"

minScore = input.int(
     4,
     "Minimum Signal Score",
     minval=1,
     maxval=6,
     group=groupFilter)

cooldownBars = input.int(
     5,
     "Signal Cooldown Bars",
     minval=0,
     maxval=50,
     group=groupFilter)

groupVisual = "🎨 Visuals"

showEMA = input.bool(
     true,
     "Show EMAs",
     group=groupVisual)

showLiquidity = input.bool(
     true,
     "Show Liquidity Sweeps",
     group=groupVisual)

showBOS = input.bool(
     true,
     "Show BOS",
     group=groupVisual)

showLevels = input.bool(
     true,
     "Show Entry / SL / TP",
     group=groupVisual)

showDashboard = input.bool(
     true,
     "Show Dashboard",
     group=groupVisual)

// ═══════════════════════════════════════════════════════════════
// CORE CALCULATIONS
// ═══════════════════════════════════════════════════════════════

atr = ta.atr(atrLen)

emaFast = ta.ema(close, emaFastLen)
emaSlow = ta.ema(close, emaSlowLen)

// Higher timeframe EMA
htfEMA = request.security(
     syminfo.tickerid,
     htfTF,
     ta.ema(close, emaSlowLen),
     barmerge.gaps_off,
     barmerge.lookahead_off)

htfClose = request.security(
     syminfo.tickerid,
     htfTF,
     close,
     barmerge.gaps_off,
     barmerge.lookahead_off)

htfBull = htfClose > htfEMA
htfBear = htfClose < htfEMA

// ═══════════════════════════════════════════════════════════════
// CANDLE STRUCTURE
// ═══════════════════════════════════════════════════════════════

bull = close > open
bear = close < open

body = math.abs(close - open)

upperWick = high - math.max(open, close)
lowerWick = math.min(open, close) - low

bodyOK = body >= atr * minBodyATR

bullPin =
     usePinBar and
     bodyOK and
     lowerWick >= body * wickRatio and
     upperWick <= body

bearPin =
     usePinBar and
     bodyOK and
     upperWick >= body * wickRatio and
     lowerWick <= body

bullEngulf =
     useEngulfing and
     bull and
     bear[1] and
     open <= close[1] and
     close >= open[1] and
     body >= body[1] * 0.9

bearEngulf =
     useEngulfing and
     bear and
     bull[1] and
     open >= close[1] and
     close <= open[1] and
     body >= body[1] * 0.9

bullPA = bullPin or bullEngulf
bearPA = bearPin or bearEngulf

// ═══════════════════════════════════════════════════════════════
// LIQUIDITY LEVELS
// ═══════════════════════════════════════════════════════════════

previousLow =
     ta.lowest(low[1], sweepLookback)

previousHigh =
     ta.highest(high[1], sweepLookback)

// Sweep below previous low,
// followed by a close back above it.
bullSweep =
     low < previousLow and
     close > previousLow and
     (previousLow - low) <= atr * sweepATR

// Sweep above previous high,
// followed by close back below it.
bearSweep =
     high > previousHigh and
     close < previousHigh and
     (high - previousHigh) <= atr * sweepATR

// ═══════════════════════════════════════════════════════════════
// MARKET STRUCTURE / BOS
// ═══════════════════════════════════════════════════════════════

swingHigh = ta.pivothigh(high, swingLen, swingLen)
swingLow  = ta.pivotlow(low, swingLen, swingLen)

var float lastSwingHigh = na
var float lastSwingLow = na

if not na(swingHigh)
    lastSwingHigh := swingHigh

if not na(swingLow)
    lastSwingLow := swingLow

bullBOS =
     not na(lastSwingHigh) and
     close > lastSwingHigh

bearBOS =
     not na(lastSwingLow) and
     close < lastSwingLow

// Alternative shorter BOS reference
recentHigh = ta.highest(high[1], bosLookback)
recentLow  = ta.lowest(low[1], bosLookback)

bullBOSFast = close > recentHigh
bearBOSFast = close < recentLow

bullStructure = bullBOS or bullBOSFast
bearStructure = bearBOS or bearBOSFast

// ═══════════════════════════════════════════════════════════════
// TREND
// ═══════════════════════════════════════════════════════════════

localBull =
     emaFast > emaSlow and
     close > emaFast

localBear =
     emaFast < emaSlow and
     close < emaFast

trendBull =
     not useTrend or
     (localBull and (not useHTF or htfBull))

trendBear =
     not useTrend or
     (localBear and (not useHTF or htfBear))

// ═══════════════════════════════════════════════════════════════
// SIGNAL SCORING
// ═══════════════════════════════════════════════════════════════

// BUY scoring
bullScore = 0

bullScore += bullSweep ? 2 : 0
bullScore += bullPA ? 1 : 0
bullScore += bullStructure ? 1 : 0
bullScore += localBull ? 1 : 0
bullScore += htfBull ? 1 : 0

// SELL scoring
bearScore = 0

bearScore += bearSweep ? 2 : 0
bearScore += bearPA ? 1 : 0
bearScore += bearStructure ? 1 : 0
bearScore += localBear ? 1 : 0
bearScore += htfBear ? 1 : 0

// ═══════════════════════════════════════════════════════════════
// FINAL CONDITIONS
// ═══════════════════════════════════════════════════════════════

var int lastSignalBar = na

cooldownOK =
     na(lastSignalBar) or
     bar_index - lastSignalBar > cooldownBars

buyLiquidityOK =
     not requireSweep or bullSweep

sellLiquidityOK =
     not requireSweep or bearSweep

buyBOSOK =
     not requireBOS or bullStructure

sellBOSOK =
     not requireBOS or bearStructure

buySignal =
     cooldownOK and
     buyLiquidityOK and
     buyBOSOK and
     bullPA and
     trendBull and
     bullScore >= minScore

sellSignal =
     cooldownOK and
     sellLiquidityOK and
     sellBOSOK and
     bearPA and
     trendBear and
     bearScore >= minScore

// ═══════════════════════════════════════════════════════════════
// ENTRY / SL / TP
// ═══════════════════════════════════════════════════════════════

var float entryPrice = na
var float stopPrice = na
var float tp1Price = na
var float tp2Price = na

var int signalDirection = 0

if buySignal

    entryPrice := close

    // Stop below sweep/candle low
    stopPrice := math.min(low, nz(previousLow, low)) - atr * slATR

    risk = entryPrice - stopPrice

    tp1Price := entryPrice + risk * tp1RR
    tp2Price := entryPrice + risk * tp2RR

    signalDirection := 1
    lastSignalBar := bar_index

if sellSignal

    entryPrice := close

    // Stop above sweep/candle high
    stopPrice := math.max(high, nz(previousHigh, high)) + atr * slATR

    risk = stopPrice - entryPrice

    tp1Price := entryPrice - risk * tp1RR
    tp2Price := entryPrice - risk * tp2RR

    signalDirection := -1
    lastSignalBar := bar_index
