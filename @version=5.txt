// ═══════════════════════════════════════════════════════════════
// CONFIRMED HTF FUNCTIONS
// ═══════════════════════════════════════════════════════════════

// IMPORTANT:
//
// [1] means we use the PREVIOUS CLOSED HTF candle.
// lookahead_on then safely carries that confirmed value
// across the lower timeframe bars.
//
// This prevents the current unfinished HTF candle from changing
// the historical/current signal.

// ───────────────────────────────────────────────────────────────
// BULL TREND
// ───────────────────────────────────────────────────────────────

f_confirmedBullTrend(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         (
             ta.ema(close, emaFastLen)[1] >
             ta.ema(close, emaSlowLen)[1] and
             close[1] >
             ta.ema(close, emaFastLen)[1]
         ),
         barmerge.gaps_off,
         barmerge.lookahead_on)

// ───────────────────────────────────────────────────────────────
// BEAR TREND
// ───────────────────────────────────────────────────────────────

f_confirmedBearTrend(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         (
             ta.ema(close, emaFastLen)[1] <
             ta.ema(close, emaSlowLen)[1] and
             close[1] <
             ta.ema(close, emaFastLen)[1]
         ),
         barmerge.gaps_off,
         barmerge.lookahead_on)


// ═══════════════════════════════════════════════════════════════
// CONFIRMED HTF STRUCTURE
// ═══════════════════════════════════════════════════════════════

f_confirmedStructureBull(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         close[1] > ta.highest(high[2], swingLen * 2),
         barmerge.gaps_off,
         barmerge.lookahead_on)

f_confirmedStructureBear(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         close[1] < ta.lowest(low[2], swingLen * 2),
         barmerge.gaps_off,
         barmerge.lookahead_on)


// ═══════════════════════════════════════════════════════════════
// CONFIRMED HTF LOCATION
// ═══════════════════════════════════════════════════════════════

f_confirmedLocationBull(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         close[1] > ta.ema(close, emaFastLen)[1],
         barmerge.gaps_off,
         barmerge.lookahead_on)

f_confirmedLocationBear(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         close[1] < ta.ema(close, emaFastLen)[1],
         barmerge.gaps_off,
         barmerge.lookahead_on)


// ═══════════════════════════════════════════════════════════════
// CONFIRMED HTF LIQUIDITY
// ═══════════════════════════════════════════════════════════════

f_confirmedSweepBull(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         low[1] < ta.lowest(low[2], liquidityLookback) and
         close[1] > ta.lowest(low[2], liquidityLookback),
         barmerge.gaps_off,
         barmerge.lookahead_on)

f_confirmedSweepBear(tf) =>
    request.security(
         syminfo.tickerid,
         tf,
         high[1] > ta.highest(high[2], liquidityLookback) and
         close[1] < ta.highest(high[2], liquidityLookback),
         barmerge.gaps_off,
         barmerge.lookahead_on)


// ═══════════════════════════════════════════════════════════════
// HIGHER-TIMEFRAME DATA
// ═══════════════════════════════════════════════════════════════

biasBull =
     f_confirmedBullTrend(biasTF)

biasBear =
     f_confirmedBearTrend(biasTF)

structureBull =
     f_confirmedStructureBull(structureTF)

structureBear =
     f_confirmedStructureBear(structureTF)

locationBull =
     f_confirmedLocationBull(locationTF)

locationBear =
     f_confirmedLocationBear(locationTF)

locationSweepBull =
     f_confirmedSweepBull(locationTF)

locationSweepBear =
     f_confirmedSweepBear(locationTF)


// ═══════════════════════════════════════════════════════════════
// CURRENT ENTRY TIMEFRAME
// ═══════════════════════════════════════════════════════════════

atr = ta.atr(atrLen)

emaFast = ta.ema(close, emaFastLen)
emaSlow = ta.ema(close, emaSlowLen)


// ═══════════════════════════════════════════════════════════════
// CURRENT CANDLE
// ═══════════════════════════════════════════════════════════════

bull = close > open
bear = close < open

body = math.abs(close - open)

upperWick =
     high - math.max(open, close)

lowerWick =
     math.min(open, close) - low

bodyOK =
     body >= atr * minBodyATR


// ═══════════════════════════════════════════════════════════════
// PIN BARS
// ═══════════════════════════════════════════════════════════════

bullPin =
     usePinBar and
     bull and
     bodyOK and
     lowerWick >= body * pinWickRatio and
     upperWick <= body

bearPin =
     usePinBar and
     bear and
     bodyOK and
     upperWick >= body * pinWickRatio and
     lowerWick <= body


// ═══════════════════════════════════════════════════════════════
// ENGULFING
// ═══════════════════════════════════════════════════════════════

bullEngulf =
     useEngulfing and
     bull and
     bear[1] and
     open <= close[1] and
     close >= open[1]

bearEngulf =
     useEngulfing and
     bear and
     bull[1] and
     open >= close[1] and
     close <= open[1]

bullPA =
     bullPin or bullEngulf

bearPA =
     bearPin or bearEngulf


// ═══════════════════════════════════════════════════════════════
// CURRENT-TF LIQUIDITY SWEEP
// ═══════════════════════════════════════════════════════════════

previousLow =
     ta.lowest(low[1], liquidityLookback)

previousHigh =
     ta.highest(high[1], liquidityLookback)

bullSweep =
     low < previousLow and
     close > previousLow and
     (previousLow - low) <= atr * sweepATR

bearSweep =
     high > previousHigh and
     close < previousHigh and
     (high - previousHigh) <= atr * sweepATR


// ═══════════════════════════════════════════════════════════════
// CURRENT-TF BOS
// ═══════════════════════════════════════════════════════════════

recentHigh =
     ta.highest(high[1], bosLookback)

recentLow =
     ta.lowest(low[1], bosLookback)

bullBOS =
     close > recentHigh

bearBOS =
     close < recentLow


// ═══════════════════════════════════════════════════════════════
// CURRENT-TF TREND
// ═══════════════════════════════════════════════════════════════

localBull =
     emaFast > emaSlow and
     close > emaFast

localBear =
     emaFast < emaSlow and
     close < emaFast


// ═══════════════════════════════════════════════════════════════
// SCORE
// ═══════════════════════════════════════════════════════════════

// Maximum = 11

bullScore = 0

bullScore += biasBull ? 2 : 0
bullScore += structureBull ? 2 : 0
bullScore += locationBull ? 1 : 0
bullScore += locationSweepBull ? 2 : 0
bullScore += bullSweep ? 1 : 0
bullScore += bullPA ? 1 : 0
bullScore += bullBOS ? 1 : 0


bearScore = 0

bearScore += biasBear ? 2 : 0
bearScore += structureBear ? 2 : 0
bearScore += locationBear ? 1 : 0
bearScore += locationSweepBear ? 2 : 0
bearScore += bearSweep ? 1 : 0
bearScore += bearPA ? 1 : 0
bearScore += bearBOS ? 1 : 0


// ═══════════════════════════════════════════════════════════════
// FINAL CONFIRMED SIGNAL
// ═══════════════════════════════════════════════════════════════

var int lastSignalBar = na

cooldownOK =
     na(lastSignalBar) or
     bar_index - lastSignalBar > cooldownBars

buySweepOK =
     not requireSweep or bullSweep

sellSweepOK =
     not requireSweep or bearSweep

buyBOSOK =
     not requireBOS or bullBOS

sellBOSOK =
     not requireBOS or bearBOS

buySignal =
     barstate.isconfirmed and
     cooldownOK and
     (not useBias or biasBull) and
     (not useStructure or structureBull) and
     (not useLocation or locationBull or locationSweepBull) and
     buySweepOK and
     buyBOSOK and
     bullPA and
     localBull and
     bullScore >= minimumScore

sellSignal =
     barstate.isconfirmed and
     cooldownOK and
     (not useBias or biasBear) and
     (not useStructure or structureBear) and
     (not useLocation or locationBear or locationSweepBear) and
     sellSweepOK and
     sellBOSOK and
     bearPA and
     localBear and
     bearScore >= minimumScore


// ═══════════════════════════════════════════════════════════════
// ENTRY / SL / TP
// ═══════════════════════════════════════════════════════════════

var float entryPrice = na
var float stopPrice = na
var float tp1Price = na
var float tp2Price = na

if buySignal

    entryPrice := close

    stopPrice :=
         low - atr * slATR

    risk =
         entryPrice - stopPrice

    tp1Price =
         entryPrice + risk * tp1RR

    tp2Price =
         entryPrice + risk * tp2RR

    lastSignalBar := bar_index


if sellSignal

    entryPrice := close

    stopPrice :=
         high + atr * slATR

    risk =
         stopPrice - entryPrice

    tp1Price =
         entryPrice - risk * tp1RR

    tp2Price =
         entryPrice - risk * tp2RR

    lastSignalBar := bar_index

         8,

         dashboard,

         9,
         "ATR",
